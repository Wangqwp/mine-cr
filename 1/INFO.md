💡 常见用法
═══════════════════════════════════════════════════════════════════

# 用法 1: 单次爬取
from stock_crawler_v2 import StockCrawler
crawler = StockCrawler()
crawler.run_once()

# 用法 2: 连续爬取
crawler.run_continuous(interval=60)  # 每分钟更新

# 用法 3: 自定义处理
data = crawler.crawl_data()
for code, info in data['数据'].items():
    print(f"{info['名称']}: {info['涨跌幅%']}%")

# 用法 4: 自定义股票
class MyCrawler(StockCrawler):
    def get_hot_stocks(self):
        return ['sh600519', 'sh601398', 'sz000333']

crawler = MyCrawler()
crawler.run_once()
